z.prop.test <- function(p1, n1, p2 = NULL, n2 = NULL, p0, 
                          alternative = "two.sided", c.level = 0.95) {

  # p1 stands for proportion of sample 1
  # n1 stands for sample 1 size
  # p2 is set to null if not specificed and it will. perform one sample z test, if specify it stands for proportion for sample 2
  # n2 is same logic as above and if specified it stands for  sample 2 size
  # p0 stands for true population proportion in one sample test
  # alternative is set to two sided and the alternative hypothesis would be p1 is different from 01, if set to less than it would be testing if p1 < p0, if set to greater than if would be testing p1 > p0
  # CI stands for confidence interval, which by default is set to 95%
  
  if (is.null(p2) || is.null(n2)) {
    # if there is no input for p2 and n2 we will proceed with a one sample z test
    if (n1 * p0 <= 5 || n1 * (1 - p0) <= 5) {
      warning("Warning: Assumption for Normal Distribution Might Not be Valid.")
    }
      # this is to ensure the validity of assuming normal distribution
    
    se_null <- sqrt(p0 * (1 - p0) / n1) # calculate standard error for proportion under null hypothesis
    
    Z <- (p1 - p0) / se_null # calculate the Z statistic which is useful later for calculating p value
    
       # Calculate p value based on input into "alternative"
    if (alternative == "two.sided") {
      # For a two-sided test
      p_value <- 2 * pnorm(-abs(Z))
    } else if (alternative == "less") {
      # For a 'less than' test: H1: p1 < p0, we use the lower tail.
      p_value <- pnorm(Z)
    } else if (alternative == "greater") {
      # For a 'greater than' test: H1: p1 > p0,  we use the upper tail.
      p_value <- pnorm(Z, lower.tail = FALSE)
    } 
    
    # calculate sample standard error for making the CI, note here we are using 1-p1, for the se_null we were using 1-p0
    se <- sqrt(p1 * (1 - p1) / n1)
    
    # find the Z(alpha/2)
    z_alpha <- qnorm(1 - (1 - c.level) / 2)
    
    # Make the upper and lower bound confidence interval for p1
    CI <- c(p1 - z_alpha * se, p1 + z_alpha * se)
    
    # Return the results as a list.
    return(list(Z = Z, P = p_value, CI = CI))
    
  } else {
    # the following is for processing the two sample test; if p2 and n2 is specify, we are testing for the difference between two proportion, the null hypothesis here is H0: p1 - p2 = 0
    
    # Check the validity of normal distribution assumption for both sample
    if (n1 * p1 <= 5 || n1 * (1 - p1) <= 5) {
      warning("Warning: Assumption for Normal Distribution Might Not be Valid.")
    }
    if (n2 * p2 <= 5 || n2 * (1 - p2) <= 5) {
      warning("Warning: Assumption for Normal Distribution Might Not be Valid.")
    }
    
    # Calculate the standard error for the difference in proportions.
    se_diff <- sqrt((p1 * (1 - p1) / n1) + (p2 * (1 - p2) / n2))
    
    # Compute the Z statistic. notice the formula is different for two sample test 
    Z <- (p1 - p2) / se_diff
    
    # Calculate p value based on input into "alternative"
    if (alternative == "two.sided") {
      p_value <- 2 * pnorm(-abs(Z))
    } else if (alternative == "less") {
      # For 'less', we test if p1 is significantly less than p2.
      p_value <- pnorm(Z)
    } else if (alternative == "greater") {
      # For 'greater', we test if p1 is significantly greater than p2.
      p_value <- 1 - pnorm(Z)
    }
    
    # creating CI for the difference in proportions
    z_alpha <- qnorm(1 - (1 - c.level) / 2)
    CI <- c((p1 - p2) - z_alpha * se_diff, (p1 - p2) + z_alpha * se_diff)
    
    return(list(Z = Z, P = p_value, CI = CI))
  }
}
